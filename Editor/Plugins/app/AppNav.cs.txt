using System;
using System.Collections.Generic;
using System.IO;
using System.Text;
using System.Text.RegularExpressions;
using Exerussus.AppCore;
using Exerussus.AppCore.Editor;
using Exerussus.AppCore.Navigation;
using Exerussus.AppCore.Views;
using UnityEditor;

namespace Exerussus.Nexus.Pages.App
{
    /// <summary>
    /// Фасад над <see cref="AppNavigation"/> (app-core): имена классов, Navigation.uss, NavTargets.cs и
    /// перепись классов в uxml пишет AppCore — страница и инструменты AppCore выдают одни и те же файлы.
    /// Своё здесь — только чтение uxml для подсчётов на вкладках.
    /// </summary>
    internal static class AppNav
    {
        /// <summary>Стандартная цель «назад» — класс есть всегда.</summary>
        public const string BackTargetId = AppNavigation.BackTargetId;

        public static string DefaultPrefix(ViewKind kind) => AppNavigation.DefaultPrefix(kind);
        public const string DefaultPostfix = AppNavigation.DefaultPostfix;

        public static string Compose(string prefix, string postfix, string id) => AppNavigation.Compose(prefix, postfix, id);

        /// <summary>Аффиксы вида из NavigationSettings; пусто — дефолт.</summary>
        public static (string prefix, string postfix) Affixes(NavigationSettings s, ViewKind kind) => AppNavigation.Affixes(s, kind);

        public static long HashOf(ViewKind kind, string id) => AppNavigation.HashOf(kind, id);

        /// <summary>Полностью перезаписывает Navigation.uss (только при изменении): класс на каждую страницу и попап + стандартные.</summary>
        public static void WriteNavigationUss(NavigationSettings s) => AppNavigation.WriteNavigationUss(s);

        /// <summary>Записи «класс → цель» для рантайма (NavigationLink).</summary>
        public static List<NavigationSettings.NavigationEntry> BuildEntries(NavigationSettings s) => AppNavigation.BuildEntries(s);

        /// <summary>Перезаписывает NavTargets.cs: PageId, PopupId и FragmentId на каждый id.</summary>
        public static bool WriteNavTargets(NavigationSettings s, out string report) => AppNavigation.WriteNavTargets(s, out report);

        public static readonly Regex ValidClassRx = AppNavigation.ValidClassRx;

        /// <summary>Единый проход: каждый токен ищется в map ровно один раз → цепочки X→Y→Z не возникают.</summary>
        public static string RewriteTokens(string text, Dictionary<string, string> map, out bool changed) =>
            AppNavigation.RewriteTokens(text, map, out changed);

        // ============================================================ uxml: классы

        // Атрибут class="..." / class='...' в uxml. Singleline — на случай переносов внутри значения.
        private static readonly Regex ClassAttrRx =
            new(@"class\s*=\s*(?<q>[""'])(?<val>.*?)\k<q>", RegexOptions.Compiled | RegexOptions.Singleline);

        private static readonly char[] WS = { ' ', '\t', '\n', '\r' };

        public static List<(string path, string text)> LoadAllUxml()
        {
            var list = new List<(string, string)>();
            foreach (var guid in AssetDatabase.FindAssets("t:VisualTreeAsset"))
            {
                var path = AssetDatabase.GUIDToAssetPath(guid);
                if (string.IsNullOrEmpty(path) || !path.EndsWith(".uxml", StringComparison.OrdinalIgnoreCase)) continue;
                if (!path.StartsWith("Assets/", StringComparison.Ordinal)) continue;
                try { list.Add((path, File.ReadAllText(path))); }
                catch { /* нечитаемый файл — пропускаем */ }
            }
            return list;
        }

        public static int CountToken(string text, string token)
        {
            var count = 0;
            foreach (Match m in ClassAttrRx.Matches(text))
                foreach (var t in m.Groups["val"].Value.Split(WS, StringSplitOptions.RemoveEmptyEntries))
                    if (string.Equals(t, token, StringComparison.Ordinal)) count++;
            return count;
        }
    }
}
