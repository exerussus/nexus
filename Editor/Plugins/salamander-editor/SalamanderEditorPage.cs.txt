using System;
using System.IO;
using Dsl.Compilation;
using Dsl.Ide;
using Dsl.Tooling;
using Dsl.Unity;
using Exerussus.Nexus.Abstractions;
using UnityEditor;
using UnityEngine;
using UnityEngine.UIElements;

namespace Exerussus.Nexus.Pages.SalamanderEditor
{
    // Страница-оболочка над рантайм-IDE Salamander (Dsl.Ide из пакета Salamander).
    // Вся IDE — редактор, подсказки, компиляция, сессия — живёт в пакете и та же
    // самая работает в игре; здесь только редакторский хост: диалоги Unity,
    // drag&drop из Project, точечный импорт сохранённых .sal, бэкап буферов перед
    // перезагрузкой домена, «Применить» в Play Mode и цвета оболочки из темы Nexus.
    // Страница обслуживающая: её ядро — вне Nexus (в пакете), модули — в проекте.
    [EditorPage("salamander-editor", "Salamander", PageCategory.Game, order: 30)]
    public sealed class SalamanderEditorPage : EditorPageBase
    {
        private VisualElement _root;
        private SalIdeView _ide;
        private PagePrefs _prefs;
        private FileIdeStorage _storage;
        private BootstrapApplyTarget _apply;
        private bool _focused;
        private double _nextApplyCheck;

        private string PrefsPath => Context.GetUserConfigPath("prefs.json");
        private string IdeStorageDir => Path.Combine(Path.GetDirectoryName(Context.GetUserConfigPath("prefs.json")) ?? ".", "ide");

        protected override void OnInitialize()
        {
            // Дёшево: только подписки (снимаются Track-ом). IDE создаётся в первом OnEnter.
            AssemblyReloadEvents.beforeAssemblyReload += OnBeforeReload;
            Track(new Unsubscribe(() => AssemblyReloadEvents.beforeAssemblyReload -= OnBeforeReload));
            EditorApplication.quitting += OnBeforeReload;
            Track(new Unsubscribe(() => EditorApplication.quitting -= OnBeforeReload));
            EditorApplication.playModeStateChanged += OnPlayModeChanged;
            Track(new Unsubscribe(() => EditorApplication.playModeStateChanged -= OnPlayModeChanged));
            // сцену могут перезагрузить прямо в Play Mode — цель «Применить» тогда
            // указывает на уничтоженный бутстрап; проверяем раз в секунду, пока страница открыта
            EditorApplication.update += OnEditorTick;
            Track(new Unsubscribe(() => EditorApplication.update -= OnEditorTick));
        }

        public override VisualElement BuildUI()
        {
            _root = new VisualElement { style = { flexGrow = 1 } };
            RegisterDragAndDrop(_root);
            return _root;
        }

        public override void OnEnter()
        {
            EnsureIde();
            // высокочастотная работа — только между OnFocus и OnUnfocus
            _ide?.SetActive(_focused);
        }

        public override void OnFocus()
        {
            _focused = true;
            _ide?.SetActive(true);
            RefreshApplyTarget();
        }

        public override void OnUnfocus()
        {
            _focused = false;
            _ide?.SetActive(false);
            _ide?.PersistSession();
        }

        public override void OnExit() => _ide?.PersistSession();

        public override void OnRefresh()
        {
            _ide?.Refresh();
            RefreshApplyTarget();
        }

        public override bool CanClose()
        {
            if (_ide == null || !_ide.HasUnsavedChanges) return true;
            int r = EditorUtility.DisplayDialogComplex("Salamander",
                "В открытых файлах Salamander есть несохранённые изменения.",
                "Сохранить все", "Отмена", "Не сохранять");
            if (r == 1) return false;
            if (r == 0)
            {
                _ide.SaveAll();
                // сохранение могло не пройти (ошибка записи, конфликт с диском) — тогда не закрываем
                return !_ide.HasUnsavedChanges;
            }
            _ide.DiscardAll();
            return true;
        }

        public override void OnPrePack()
        {
            _ide?.PersistSession();
            _prefs?.Save(PrefsPath);
        }

        protected override void OnDisposing()
        {
            _apply?.Dispose();
            _apply = null;
            _ide?.Dispose();
            _ide = null;
        }

        // ── IDE ────────────────────────────────────────────────────────────────

        private void EnsureIde()
        {
            if (_ide != null || _root == null) return;
            _prefs ??= PagePrefs.Load(PrefsPath);
            _storage = new FileIdeStorage(IdeStorageDir,
                (what, e) => NexusDiagnostics.Swallowed("salamander-editor/storage " + what, e));

            string modulesRoot = ResolveModulesRoot();
            _ide = new SalIdeView(new IdeOptions
            {
                Host = new EditorIdeHost((text, kind) => Context.SetStatus(text, kind)),
                Storage = _storage,
                Workspace = new FileSystemWorkspace(modulesRoot),
                Api = new ManifestFileApiSource(ResolveApiPath(modulesRoot), _storage),
            });
            _ide.ApplyChrome(ChromeFromTheme());

            var extras = _ide.ToolbarExtras;
            extras.Add(ToolbarButton("Открыть…", "Открыть .sal с диска (можно и перетащить файл на страницу)", OpenFileDialog));
            extras.Add(ToolbarButton("Папка модулей…", "Где искать модули (папки с module.json)", PickModulesRoot));
            extras.Add(ToolbarButton("API-манифест…", "salamander-api.json игры — события и API для подсказок и проверки", PickManifest));

            // миграция v1: последний открытый файл — если сессии ещё нет
            if (!string.IsNullOrEmpty(_prefs.LastFilePath))
            {
                if (_ide.Documents.Count == 0 && File.Exists(_prefs.LastFilePath)) _ide.OpenFile(_prefs.LastFilePath);
                _prefs.LastFilePath = null;
                _prefs.Save(PrefsPath);
            }

            _root.Add(_ide);
        }

        private Button ToolbarButton(string text, string hint, Action onClick)
        {
            var b = Context.Ui.Button(NexusRole.Common, text, onClick);
            b.tooltip = hint;
            b.style.height = 22;
            b.style.marginLeft = 2;
            b.style.marginRight = 2;
            return b;
        }

        // Оболочка IDE — токенами палитры Nexus (кодовая область — своя палитра в USS пакета).
        private IdeChromeColors ChromeFromTheme()
        {
            var t = Context.Theme;
            return new IdeChromeColors
            {
                Background = t.Get(NexusToken.BgHard),
                Panel = t.Get(NexusToken.BgSoft),
                Raised = t.Get(NexusToken.BgRaised),
                Border = t.Get(NexusToken.Divider),
                Text = t.Get(NexusToken.TextNormal),
                TextDim = t.Get(NexusToken.TextDim),
                Accent = t.Get(NexusToken.Accent),
                AccentText = t.Get(NexusToken.OnAccent),
                Selection = t.Get(NexusToken.Selection),
                Hover = t.Get(NexusToken.SurfaceHover),
                Error = t.Get(NexusToken.Error),
                Warning = t.Get(NexusToken.Warning),
                Ok = t.Get(NexusToken.Ok),
            };
        }

        // Папка модулей: выбранная, иначе StreamingAssets (если там есть модули), иначе Assets.
        private string ResolveModulesRoot()
        {
            if (!string.IsNullOrEmpty(_prefs.ModulesRoot) && Directory.Exists(_prefs.ModulesRoot)) return _prefs.ModulesRoot;
            string sa = Application.streamingAssetsPath;
            if (Directory.Exists(sa) && WorkspaceLoader.FindModuleDirs(sa).Count > 0) return sa;
            return Application.dataPath;
        }

        // Манифест: выбранный, иначе рядом с модулями, иначе где-нибудь в Assets.
        private string ResolveApiPath(string modulesRoot)
        {
            if (!string.IsNullOrEmpty(_prefs.ApiManifestPath)) return _prefs.ApiManifestPath;
            return WorkspaceLoader.FindApiManifest(modulesRoot)
                   ?? ModuleLoader.FindNearestFile(Application.dataPath, WorkspaceLoader.ApiManifestFileName);
        }

        private void OpenFileDialog()
        {
            string dir = _ide?.Workspace?.Root ?? Application.dataPath;
            string p = EditorUtility.OpenFilePanel("Открыть .sal", dir, "sal");
            if (!string.IsNullOrEmpty(p)) _ide?.OpenFile(p);
        }

        private void PickModulesRoot()
        {
            string p = EditorUtility.OpenFolderPanel("Папка модулей Salamander", _ide?.Workspace?.Root ?? Application.dataPath, "");
            if (string.IsNullOrEmpty(p)) return;
            _prefs.ModulesRoot = p;
            _prefs.Save(PrefsPath);
            _ide.SetWorkspace(new FileSystemWorkspace(p));
            if (string.IsNullOrEmpty(_prefs.ApiManifestPath))
                _ide.SetApiSource(new ManifestFileApiSource(ResolveApiPath(p), _storage));
        }

        private void PickManifest()
        {
            string p = EditorUtility.OpenFilePanel("salamander-api.json", _ide?.Workspace?.Root ?? Application.streamingAssetsPath, "json");
            if (!string.IsNullOrEmpty(p)) UseManifest(p);
        }

        private void UseManifest(string path)
        {
            var src = new ManifestFileApiSource(path, _storage);
            if (src.Current.IsEmpty)
            {
                Context.SetStatus("API-манифест не принят: " + (src.Error ?? "пустой"), StatusKind.Error);
                return;
            }
            _prefs.ApiManifestPath = path;
            _prefs.Save(PrefsPath);
            _ide.SetApiSource(src);
            Context.SetStatus($"API-манифест: {Path.GetFileName(path)} (v{src.Current.ApiVersion}).", StatusKind.Ok);
        }

        // ── Play Mode: «Применить в игре» ──────────────────────────────────────

        private void OnPlayModeChanged(PlayModeStateChange change)
        {
            if (change == PlayModeStateChange.EnteredPlayMode) EditorApplication.delayCall += RefreshApplyTarget;
            else if (change == PlayModeStateChange.ExitingPlayMode) DropApplyTarget();
        }

        private void OnEditorTick()
        {
            if (!_focused || _ide == null) return;
            double now = EditorApplication.timeSinceStartup;
            if (now < _nextApplyCheck) return;
            _nextApplyCheck = now + 1.0;
            RefreshApplyTarget();
        }

        private void RefreshApplyTarget()
        {
            if (_ide == null) return;
            if (!EditorApplication.isPlaying) { DropApplyTarget(); return; }
            if (_apply != null && _apply.IsAlive) return;
            // сцену перезагрузили в Play Mode — прежний бутстрап уничтожен, ищем новый
            DropApplyTarget();
            var bootstrap = UnityEngine.Object.FindAnyObjectByType<ScriptHostBootstrap>();
            if (bootstrap == null) return;
            _apply = new BootstrapApplyTarget(bootstrap, () => _ide?.Workspace, (kind, text) => _ide?.Log(kind, text));
            _ide.SetApplyTarget(_apply);
        }

        private void DropApplyTarget()
        {
            if (_apply == null) return;
            _apply.Dispose();
            _apply = null;
            _ide?.SetApplyTarget(null);
        }

        // ── перезагрузка домена ────────────────────────────────────────────────

        // Буферы и вкладки — в UserSettings (hot-exit), фоновая компиляция — прервана:
        // живой поток в пользовательской сборке не даёт Unity выгрузить домен.
        private void OnBeforeReload() => _ide?.PrepareForShutdown();

        // ── drag&drop: .sal — открыть, .json — API-манифест ────────────────────

        private void RegisterDragAndDrop(VisualElement target)
        {
            target.RegisterCallback<DragUpdatedEvent>(_ =>
            {
                // visualMode обязан выставляться на КАЖДОМ Update, иначе редактор покажет «запрещено»
                DragAndDrop.visualMode = ScanDrag(out var _, out var _) ? DragAndDropVisualMode.Copy : DragAndDropVisualMode.Rejected;
            });
            target.RegisterCallback<DragPerformEvent>(_ =>
            {
                if (!ScanDrag(out string sal, out string json)) return;
                DragAndDrop.AcceptDrag();
                // порядок важен: сначала манифест — .sal сразу компилируется против нового API
                if (json != null) UseManifest(Path.GetFullPath(json));
                if (sal != null) _ide?.OpenFile(Path.GetFullPath(sal));
            });
        }

        private static bool ScanDrag(out string sal, out string json)
        {
            sal = null;
            json = null;
            var paths = DragAndDrop.paths;
            if (paths == null) return false;
            foreach (var p in paths)
            {
                if (string.IsNullOrEmpty(p)) continue;
                if (sal == null && p.EndsWith(".sal", StringComparison.OrdinalIgnoreCase)) sal = p;
                else if (json == null && p.EndsWith(".json", StringComparison.OrdinalIgnoreCase)) json = p;
            }
            return sal != null || json != null;
        }

        private sealed class Unsubscribe : IDisposable
        {
            private Action _action;
            public Unsubscribe(Action action) => _action = action;
            public void Dispose()
            {
                _action?.Invoke();
                _action = null;
            }
        }
    }
}
