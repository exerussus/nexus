// ============================================================
//  BuildTab.Pipeline.cs — подвкладка «Пайплайн»: пре-/пост-процессы по профилю.
//
//  У каждого профиля свой список шагов: порядок, вкл/выкл, настройки (JSON).
//  Новые шаги из кода дописываются в конец выключенными. Запись о шаге, которого больше
//  нет в коде, не удаляется молча — показывается с кнопкой «Удалить», а включённая такая
//  запись остановит сборку с понятной ошибкой.
// ============================================================

using System.Collections.Generic;
using Exerussus.Nexus.Abstractions;
using UnityEngine.UIElements;

namespace Exerussus.Nexus.Pages.App
{
    internal sealed partial class BuildTab
    {
        private void RenderPipelineTab()
        {
            var head = _kit.Card("Пайплайн профиля",
                "Пре-процессы выполняются после активации профиля, до билда; пост-процессы — только после успешного билда. " +
                "Новый шаг — класс BuildStep<T> с атрибутом [BuildStep] в коде страницы App.");
            head.Add(_kit.Labeled("Профиль", ProfileDropdown()));
            _content.Add(head);

            var selected = SelectedProfile();
            if (selected == null) return;

            var all = BuildStepRegistry.All();
            var pipeline = _config.GetOrCreate(selected.Key);
            if (BuildStepRegistry.Sync(pipeline, all)) RequestConfigSave();

            RenderStage(pipeline, all, BuildStage.Pre, "Пре-процессы");
            RenderStage(pipeline, all, BuildStage.Post, "Пост-процессы");
            RenderMissingSteps(pipeline, all);
        }

        private void RenderStage(ProfilePipeline pipeline, List<StepDescriptor> all, BuildStage stage, string title)
        {
            var card = _kit.Card(title);
            var any = false;

            foreach (var entry in pipeline.steps)
            {
                var d = BuildStepRegistry.Find(all, entry.id);
                if (d == null || d.Meta.Stage != stage) continue;

                if (any) card.Add(_kit.Divider(4f));
                any = true;
                card.Add(StepRow(pipeline, all, entry, d));
            }

            if (!any) card.Add(_kit.Faint("Шагов этого типа пока нет."));
            _content.Add(card);
        }

        private VisualElement StepRow(ProfilePipeline pipeline, List<StepDescriptor> all, StepEntry entry, StepDescriptor d)
        {
            var box = new VisualElement();
            box.style.paddingTop = 4f;
            box.style.paddingBottom = 4f;

            var row = _kit.Row();
            row.style.flexWrap = Wrap.NoWrap;

            var toggle = _kit.Check(d.Meta.Name, entry.enabled, v =>
            {
                entry.enabled = v;
                RequestConfigSave();
                Render();
            });
            toggle.style.flexGrow = 1;
            row.Add(toggle);

            row.Add(_kit.Btn(NexusRole.Ghost, "↑", () => MoveStep(pipeline, all, entry, -1), small: true));
            var down = _kit.Btn(NexusRole.Ghost, "↓", () => MoveStep(pipeline, all, entry, +1), small: true);
            down.style.marginRight = 0f;
            row.Add(down);
            box.Add(row);

            if (!string.IsNullOrEmpty(d.Meta.Description))
            {
                var desc = _kit.Hint(d.Meta.Description);
                desc.style.marginLeft = 20f;
                box.Add(desc);
            }

            if (!entry.enabled) return box;

            var settings = d.Step.LoadSettings(entry.settingsJson);
            VisualElement ui = null;
            try
            {
                ui = d.Step.BuildSettingsUI(settings, () =>
                {
                    entry.settingsJson = d.Step.SaveSettings(settings);
                    RequestConfigSave();
                }, _kit);
            }
            catch (System.Exception ex)
            {
                NexusDiagnostics.Error($"App/Сборка: UI шага {entry.id}", ex);
                box.Add(_kit.Status(StatusKind.Error, $"UI настроек шага упал: {ex.Message}"));
            }

            if (ui != null)
            {
                ui.style.marginLeft = 20f;
                ui.style.marginTop = 6f;
                _kit.PaintLabels(ui);
                _kit.PaintFieldInput(ui);
                box.Add(ui);
            }

            return box;
        }

        /// <summary>Сдвинуть шаг среди шагов той же стадии (порядок хранится в общем списке).</summary>
        private void MoveStep(ProfilePipeline pipeline, List<StepDescriptor> all, StepEntry entry, int dir)
        {
            var stage = BuildStepRegistry.Find(all, entry.id)?.Meta.Stage;
            var index = pipeline.steps.IndexOf(entry);
            if (stage == null || index < 0) return;

            for (var j = index + dir; j >= 0 && j < pipeline.steps.Count; j += dir)
            {
                var other = BuildStepRegistry.Find(all, pipeline.steps[j].id);
                if (other == null || other.Meta.Stage != stage) continue;
                pipeline.steps[index] = pipeline.steps[j];
                pipeline.steps[j] = entry;
                RequestConfigSave();
                Render();
                return;
            }
        }

        private void RenderMissingSteps(ProfilePipeline pipeline, List<StepDescriptor> all)
        {
            var missing = new List<StepEntry>();
            foreach (var e in pipeline.steps)
                if (BuildStepRegistry.Find(all, e.id) == null) missing.Add(e);
            if (missing.Count == 0) return;

            var card = _kit.Card("Шаги не найдены в коде",
                "Записи сохранены в конфиге, но класса шага нет. Включённый ненайденный шаг остановит сборку.");

            foreach (var e in missing)
            {
                var row = _kit.ListRow();
                row.Add(_kit.Dot(e.enabled ? NexusToken.Error : NexusToken.Warning));
                var id = _kit.Mono(e.id);
                id.style.flexGrow = 1;
                row.Add(id);

                var captured = e;
                var remove = _kit.Btn(NexusRole.Alarm, "Удалить", () =>
                {
                    pipeline.steps.Remove(captured);
                    RequestConfigSave();
                    Render();
                }, small: true);
                remove.style.marginRight = 0f;
                row.Add(remove);
                card.Add(row);
            }

            _content.Add(card);
        }
    }
}
