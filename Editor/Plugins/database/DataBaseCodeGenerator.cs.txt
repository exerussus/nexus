using System;
using System.Collections.Generic;
using System.IO;
using System.Text;
using UnityEditor;
using UnityEngine;

namespace Exerussus.Nexus.Pages.DataBaseEditor
{
    public static class DataBaseCodeGenerator
    {
        public static bool Generate(IReadOnlyList<DataBaseScanEntry> entries, AutoGenSettings config)
        {
            if (config == null) { Debug.LogError("[DB AutoGen] Config is null."); return false; }
            if (entries == null) entries = Array.Empty<DataBaseScanEntry>();

            var sb = new StringBuilder(4096);
            EmitFile(sb, entries, config);

            try
            {
                var dir = Path.GetDirectoryName(config.GeneratedFilePath);
                if (!string.IsNullOrEmpty(dir) && !Directory.Exists(dir))
                    Directory.CreateDirectory(dir);

                var newText = sb.ToString();
                var existingText = File.Exists(config.GeneratedFilePath)
                    ? File.ReadAllText(config.GeneratedFilePath)
                    : null;

                // Avoid touching the file (and triggering a recompile) when nothing changed.
                if (existingText == newText)
                {
                    if (config.LogActions)
                        Debug.Log($"[DB AutoGen] {config.GeneratedFilePath} is up to date.");
                    return true;
                }

                File.WriteAllText(config.GeneratedFilePath, newText);
                AssetDatabase.ImportAsset(config.GeneratedFilePath, ImportAssetOptions.ForceUpdate);

                if (config.LogActions)
                    Debug.Log($"[DB AutoGen] Regenerated {config.GeneratedFilePath} with {entries.Count} entries.");
                return true;
            }
            catch (Exception ex)
            {
                Debug.LogError($"[DB AutoGen] Failed to write {config.GeneratedFilePath}: {ex}");
                return false;
            }
        }

        // ---------- emission ----------

        private static void EmitFile(StringBuilder sb, IReadOnlyList<DataBaseScanEntry> entries, AutoGenSettings config)
        {
            sb.AppendLine("// <auto-generated>");
            sb.AppendLine("//   This file is generated by Nexus DataBase page (servicing auto-gen).");
            sb.AppendLine("//   Do not edit by hand — your changes will be overwritten.");
            sb.AppendLine("// </auto-generated>");
            sb.AppendLine();

            // Collect namespaces needed (each entry's type namespace + UnityEngine).
            var usings = new SortedSet<string>(StringComparer.Ordinal) { "UnityEngine" };
            foreach (var e in entries)
                if (!string.IsNullOrEmpty(e.Type.Namespace))
                    usings.Add(e.Type.Namespace);

            foreach (var u in usings)
                sb.Append("using ").Append(u).AppendLine(";");
            sb.AppendLine();

            var hasNs = !string.IsNullOrWhiteSpace(config.GeneratedNamespace);
            var indent = hasNs ? "    " : "";

            if (hasNs)
            {
                sb.Append("namespace ").Append(config.GeneratedNamespace).AppendLine();
                sb.AppendLine("{");
            }

            sb.Append(indent).AppendLine("public static partial class DataBase");
            sb.Append(indent).AppendLine("{");

            // Fields
            foreach (var e in entries)
            {
                sb.Append(indent).Append("    private static ")
                  .Append(e.Type.Name).Append(' ')
                  .Append(FieldName(e.Alias, config.FieldPrefix)).AppendLine(";");
            }
            if (entries.Count > 0) sb.AppendLine();

            // Properties
            foreach (var e in entries)
            {
                var field = FieldName(e.Alias, config.FieldPrefix);
                if (config.DebuggerStepThrough)
                    sb.Append(indent).AppendLine("    [System.Diagnostics.DebuggerStepThrough]");
                sb.Append(indent).Append("    public static ").Append(e.Type.Name).Append(' ').Append(e.Alias).AppendLine();
                sb.Append(indent).AppendLine("    {");
                sb.Append(indent).AppendLine("        get");
                sb.Append(indent).AppendLine("        {");
                if (config.UseUnityEditorGuard)
                    sb.Append(indent).AppendLine("#if UNITY_EDITOR");
                sb.Append(indent).Append("            DataBaseBuilder.TryInit(ref ").Append(field)
                  .Append(", \"").Append(e.AssetPath).AppendLine("\");");
                sb.Append(indent).Append("            DataBaseBuilder.RegisterDropdown<").Append(e.Alias).Append("DropdownAttribute>(")
                  .Append(field).AppendLine(");");
                if (config.UseUnityEditorGuard)
                    sb.Append(indent).AppendLine("#endif");
                sb.Append(indent).Append("            return ").Append(field).AppendLine(";");
                sb.Append(indent).AppendLine("        }");
                sb.Append(indent).AppendLine("    }");
                sb.AppendLine();
            }

            // InitializeGenerated — called from DataBase.Initialize() via reflection.
            sb.Append(indent).AppendLine("[RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.BeforeSceneLoad)]");
            sb.Append(indent).AppendLine("    public static void Initialize()");
            sb.Append(indent).AppendLine("    {");
            foreach (var e in entries)
            {
                var field = FieldName(e.Alias, config.FieldPrefix);
                var resourceName = System.IO.Path.GetFileNameWithoutExtension(e.AssetPath);
                sb.Append(indent).Append("        ").Append(field)
                  .Append(" = Resources.Load<").Append(e.Type.Name).Append(">(\"")
                  .Append(resourceName).AppendLine("\");");
                sb.Append(indent).Append("        if (").Append(field)
                  .Append(" != null) ").Append(field).AppendLine(".Initialize();");
            }
            sb.Append(indent).AppendLine("    }");

            sb.Append(indent).AppendLine("}");

            // Per-alias dropdown attribute classes.
            if (entries.Count > 0)
            {
                sb.AppendLine();
                foreach (var e in entries)
                {
                    sb.Append(indent)
                      .Append("public sealed class ").Append(e.Alias)
                      .AppendLine("DropdownAttribute : BaseDropdownAttribute { }");
                }
            }

            if (hasNs) sb.AppendLine("}");
        }

        private static string FieldName(string alias, string prefix)
        {
            if (string.IsNullOrEmpty(alias)) return prefix + "field";
            return prefix + char.ToLowerInvariant(alias[0]) + alias.Substring(1);
        }
    }
}
