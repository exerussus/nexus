// ============================================================
//  AppPage.Generation.cs — вкладка «Generation».
//
//  Блоки:
//    • "Generate PageId / PopupId" — по id из NavigationData полностью перезаписывает
//      NavTargets.cs (PageId на страницы, PopupId на попапы). [Переименовано из
//      NavigationLink, чтобы не конфликтовать с рантайм-классом роутинга.]
//    • Page & popup templates — копия Default.uxml в Assets/App/Pages|Popups/<Class>/<id>.uxml.
//    • App scaffold — НОВОЕ: идемпотентная генерация дефолтного следа проекта
//      (контейнеры с mainContainer+заливкой и общим AppStyle, AppStyle.uss, AppTheme.tss,
//      App Settings (PanelSettings), UISoundLibrary, NavigationData, NavigationSettings).
//      Создаём только отсутствующее — авторские правки никогда не перетираем.
//
//  Правила именования (осознанно разные):
//    • id -> идентификатор типа: '_' и '-' — разделители, отбрасываются: game_over -> GameOver
//    • имя элемента -> const: '-' отбрасывается, '_' СОХРАНЯЕТСЯ: restart-button_easy -> RestartButton_Easy
//
//  Все пути берутся из AppPaths (проектный конфиг). Реестр id — NavigationData (db больше нет).
//  Хелперы рефлексии сцены (ResolveType/ReadStringField/ReadVisualTree) лежат ЗДЕСЬ и
//  переиспользуются графом — не дублировать их в AppPage.Graph.cs.
// ============================================================

        // Пересобирает список: для каждого id из NavigationData проверяем папку и uxml внутри.
        private void RefreshTemplates()
        {
            if (_tplList == null) return;
            _tplList.Clear();

            var pageIds  = Distinct(_pageReg.ReadStored());
            var popupIds = Distinct(_popupReg.ReadStored());

            bool defaultExists = File.Exists(AppPaths.DefaultTemplatePath);
            int total = 0, missing = 0;

            foreach (var id in pageIds)  { if (AddTemplateRow(id, false)) missing++; total++; }
            foreach (var id in popupIds) { if (AddTemplateRow(id, true))  missing++; total++; }

            var status = new StringBuilder();
            if (!defaultExists)
                status.Append($"(!) Default template not found at {AppPaths.DefaultTemplatePath} (scaffold it below). ");
            if (total == 0)
                status.Append("No page/popup ids in NavigationData.");
            else
                status.Append($"{total} item(s), {missing} missing template(s).");
            _tplStatus.text = status.ToString();

            _tplGenAllBtn?.SetEnabled(missing > 0);
            ApplyTheme();
        }

        // Возвращает true, если шаблон отсутствует (строка «missing»).
        private bool AddTemplateRow(string id, bool isPopup)
        {
            var folder   = TemplateFolder(id, isPopup);
            var uxmlPath = TemplateUxmlPath(id, isPopup);

            bool folderOk = AssetDatabase.IsValidFolder(folder);
            bool uxmlOk   = File.Exists(uxmlPath);
            var state = !folderOk ? TplState.MissingFolder : (!uxmlOk ? TplState.MissingUxml : TplState.Ok);

            var row = new VisualElement();
            row.AddToClassList("nav-tpl-row");

            var idLabel = new Label((isPopup ? "[popup] " : "[page] ") + id);
            idLabel.AddToClassList("nav-tpl-id");

            var pathLabel = new Label(uxmlPath);
            pathLabel.AddToClassList("nav-tpl-path");

            var spacer = new VisualElement();
            spacer.AddToClassList("nav-spacer");

            var stateLabel = new Label(StateText(state));
            stateLabel.AddToClassList("nav-tpl-state");
            stateLabel.AddToClassList(state == TplState.Ok ? "nav-tpl-state--ok" : "nav-tpl-state--missing");

            row.Add(idLabel);
            row.Add(pathLabel);
            row.Add(spacer);
            row.Add(stateLabel);

            if (state != TplState.Ok)
            {
                var capturedId = id;
                var capturedPopup = isPopup;
                var genBtn = new Button(() => GenerateOneTemplate(capturedId, capturedPopup)) { text = "Generate template" };
                genBtn.AddToClassList("nav-btn");
                genBtn.AddToClassList("nav-tpl-gen");
                row.Add(genBtn);
            }
            else
            {
                var capturedPath = uxmlPath;
                row.AddToClassList("nav-tpl-row--pingable");
                row.tooltip = "Click to ping in Project: " + capturedPath;
                row.RegisterCallback<PointerDownEvent>(e =>
                {
                    if (e.button != 0) return;
                    var asset = AssetDatabase.LoadAssetAtPath<UnityEngine.Object>(capturedPath);
                    if (asset != null)
                    {
                        EditorGUIUtility.PingObject(asset);
                        Selection.activeObject = asset;
                    }
                });
            }

            _tplList.Add(row);
            return state != TplState.Ok;
        }

        private static string StateText(TplState s) => s switch
        {
            TplState.Ok            => "OK",
            TplState.MissingUxml   => "no uxml",
            TplState.MissingFolder => "no folder",
            _                      => "?"
        };

        private void GenerateOneTemplate(string id, bool isPopup)
        {
            bool ok = CopyTemplate(id, isPopup, out var msg);
            AssetDatabase.Refresh();
            RefreshTemplates();
            SetGenLog("Template: " + msg, ok ? StatusKind.Ok : StatusKind.Error);
        }

        private void GenerateAllTemplates()
        {
            if (!File.Exists(AppPaths.DefaultTemplatePath))
            {
                SetGenLog($"Default template not found at {AppPaths.DefaultTemplatePath} — scaffold it first.", StatusKind.Error);
                return;
            }

            var pageIds  = Distinct(_pageReg.ReadStored());
            var popupIds = Distinct(_popupReg.ReadStored());

            int created = 0, failed = 0;
            var sb = new StringBuilder();

            foreach (var id in pageIds)  Run(id, false);
            foreach (var id in popupIds) Run(id, true);

            AssetDatabase.Refresh();
            RefreshTemplates();

            sb.Insert(0, $"Templates: created {created}" + (failed > 0 ? $", {failed} failed" : "") +
                         (created == 0 && failed == 0 ? " (nothing missing)" : "") + ".\n");
            SetGenLog(sb.ToString().TrimEnd(), failed > 0 ? StatusKind.Warning : StatusKind.Ok);

            void Run(string id, bool isPopup)
            {
                if (File.Exists(TemplateUxmlPath(id, isPopup))) return;   // уже есть — пропускаем
                if (CopyTemplate(id, isPopup, out var msg)) { created++; sb.AppendLine("+ " + msg); }
                else                                         { failed++;  sb.AppendLine("! " + msg); }
            }
        }

        // Копирует Default.uxml в папку страницы/попапа под именем <id>.uxml.
        private static bool CopyTemplate(string id, bool isPopup, out string msg)
        {
            if (string.IsNullOrWhiteSpace(id)) { msg = "empty id — skipped"; return false; }

            if (!File.Exists(AppPaths.DefaultTemplatePath))
            {
                msg = $"Default template not found at {AppPaths.DefaultTemplatePath}";
                return false;
            }

            var folder = TemplateFolder(id, isPopup);
            var dest   = TemplateUxmlPath(id, isPopup);

            if (File.Exists(dest)) { msg = $"already exists: {dest}"; return true; }

            try { EnsureAssetFolder(folder); }
            catch (Exception e) { msg = $"cannot create folder {folder}: {e.Message}"; return false; }

            var copied = AssetDatabase.CopyAsset(AppPaths.DefaultTemplatePath, dest);
            if (!copied) { msg = $"CopyAsset failed -> {dest}"; return false; }

            msg = dest;
            return true;
        }

        // Папка = root/<PascalCase(id)>; имя файла = сырой id, почищенный от запретных символов.
        private static string TemplateFolder(string id, bool isPopup) =>
            $"{(isPopup ? AppPaths.PopupsRoot : AppPaths.PagesRoot)}/{ToTypeIdentNonEmpty(id, isPopup ? "Popup" : "Page")}";

        private static string TemplateUxmlPath(string id, bool isPopup) =>
            $"{TemplateFolder(id, isPopup)}/{SanitizeFileName(id)}.uxml";

        private static string SanitizeFileName(string id)
        {
            id = (id ?? string.Empty).Trim();
            if (id.Length == 0) return "page";
            var invalid = Path.GetInvalidFileNameChars();
            var sb = new StringBuilder(id.Length);
            foreach (var ch in id) sb.Append(Array.IndexOf(invalid, ch) >= 0 ? '_' : ch);
            return sb.ToString();
        }

        // Рекурсивно создаёт папку ассета через AssetDatabase (с .meta). "Assets" всегда валиден.
        private static void EnsureAssetFolder(string folder)
        {
            folder = folder.Replace('\\', '/').TrimEnd('/');
            if (string.IsNullOrEmpty(folder) || AssetDatabase.IsValidFolder(folder)) return;

            var parent = Path.GetDirectoryName(folder)?.Replace('\\', '/');
            var leaf   = Path.GetFileName(folder);
            if (string.IsNullOrEmpty(parent) || string.IsNullOrEmpty(leaf)) return;

            EnsureAssetFolder(parent);
            AssetDatabase.CreateFolder(parent, leaf);
        }

        private void RefreshScaffold()
        {
            if (_scaffoldList == null) return;
            _scaffoldList.Clear();

            int present = 0, total = 0;
            foreach (var item in ScaffoldItems())
            {
                bool exists = ScaffoldExists(item.Path);
                if (exists) present++;
                total++;
                _scaffoldList.Add(MakeScaffoldRow(item, exists));
            }

            int missing = total - present;
            _scaffoldGenAllBtn?.SetEnabled(missing > 0);
            if (_scaffoldStatus != null)
                _scaffoldStatus.text = $"{present}/{total} present, {missing} missing.";
            ApplyTheme();
        }

        private static bool ScaffoldExists(string path) =>
            AssetDatabase.LoadAssetAtPath<UnityEngine.Object>(path) != null || File.Exists(path);

        private VisualElement MakeScaffoldRow(ScaffoldItem item, bool exists)
        {
            var row = new VisualElement();
            row.AddToClassList("nav-tpl-row");

            var idLabel = new Label(item.Label);
            idLabel.AddToClassList("nav-tpl-id");

            var pathLabel = new Label(item.Path);
            pathLabel.AddToClassList("nav-tpl-path");

            var spacer = new VisualElement();
            spacer.AddToClassList("nav-spacer");

            var stateLabel = new Label(exists ? "present" : "missing");
            stateLabel.AddToClassList("nav-tpl-state");
            stateLabel.AddToClassList(exists ? "nav-tpl-state--ok" : "nav-tpl-state--missing");

            row.Add(idLabel);
            row.Add(pathLabel);
            row.Add(spacer);
            row.Add(stateLabel);

            if (!exists)
            {
                var kind = item.Kind;
                var genBtn = new Button(() => ScaffoldOneAndRefresh(kind)) { text = "Create" };
                genBtn.AddToClassList("nav-btn");
                genBtn.AddToClassList("nav-tpl-gen");
                row.Add(genBtn);
            }
            else
            {
                var capturedPath = item.Path;
                row.AddToClassList("nav-tpl-row--pingable");
                row.RegisterCallback<PointerDownEvent>(e =>
                {
                    if (e.button != 0) return;
                    var obj = AssetDatabase.LoadAssetAtPath<UnityEngine.Object>(capturedPath);
                    if (obj != null) EditorGUIUtility.PingObject(obj);
                });
            }

            return row;
        }

        private void ScaffoldOneAndRefresh(ScaffoldKind kind)
        {
            bool ok = ScaffoldOne(kind, out var msg);
            AssetDatabase.SaveAssets();
            AssetDatabase.Refresh();
            RefreshScaffold();
            RefreshTemplates();
            SetGenLog("Scaffold: " + msg, ok ? StatusKind.Ok : StatusKind.Error);
        }

        private void ScaffoldAllMissing()
        {
            int created = 0, failed = 0, skipped = 0;
            var sb = new StringBuilder();

            // Порядок важен: стиль и тема — до App Settings (PanelSettings линкует тему).
            foreach (var item in ScaffoldItems())
            {
                if (ScaffoldExists(item.Path)) { skipped++; continue; }
                if (ScaffoldOne(item.Kind, out var msg)) { created++; sb.AppendLine("+ " + msg); }
                else                                      { failed++;  sb.AppendLine("! " + msg); }
            }

            AssetDatabase.SaveAssets();
            AssetDatabase.Refresh();
            RefreshScaffold();
            RefreshTemplates();

            sb.Insert(0, $"Scaffold: created {created}, skipped {skipped}" +
                         (failed > 0 ? $", {failed} failed" : "") + ".\n");
            SetGenLog(sb.ToString().TrimEnd(), failed > 0 ? StatusKind.Warning : StatusKind.Ok);
        }

        // Идемпотентное создание одного артефакта. Существующее не трогаем.
        private bool ScaffoldOne(ScaffoldKind kind, out string msg)
        {
            try
            {
                switch (kind)
                {
                    case ScaffoldKind.AppStyle:
                        return WriteTextAssetIfMissing(AppPaths.AppStylePath, AppStyleContent(), out msg);

                    case ScaffoldKind.AppTheme:
                        return WriteTextAssetIfMissing(AppPaths.AppThemePath, AppThemeContent(), out msg);

                    case ScaffoldKind.AppSettings:
                        return CreatePanelSettings(out msg);

                    case ScaffoldKind.AppStarter:
                        return WriteTextAssetIfMissing(AppPaths.AppStarterPath, ContainerUxmlContent(), out msg);

                    case ScaffoldKind.DefaultContainer:
                        return WriteTextAssetIfMissing(AppPaths.DefaultTemplatePath, ContainerUxmlContent(), out msg);

                    case ScaffoldKind.LoadingScreen:
                        return WriteTextAssetIfMissing(AppPaths.LoadingScreenPath, ContainerUxmlContent(), out msg);

                    case ScaffoldKind.UISoundLibrary:
                        return CreateScriptableIfMissing<UISoundLibrary>(AppPaths.UISoundLibraryPath, out msg);

                    case ScaffoldKind.NavigationData:
                        return CreateScriptableIfMissing<NavigationData>(AppPaths.NavigationDataPath, out msg);

                    case ScaffoldKind.NavigationSettings:
                        return CreateScriptableIfMissing<NavigationSettings>(AppPaths.SettingPath, out msg);

                    default:
                        msg = "unknown kind"; return false;
                }
            }
            catch (Exception e)
            {
                msg = $"{kind}: {e.Message}";
                return false;
            }
        }

        // ---- writers ----

        private static bool WriteTextAssetIfMissing(string path, string content, out string msg)
        {
            if (ScaffoldExists(path)) { msg = $"already exists: {path}"; return true; }

            var dir = Path.GetDirectoryName(path)?.Replace('\\', '/');
            if (!string.IsNullOrEmpty(dir)) EnsureAssetFolder(dir);

            File.WriteAllText(path, content, new UTF8Encoding(false));
            AssetDatabase.ImportAsset(path);
            msg = path;
            return true;
        }

        private static bool CreateScriptableIfMissing<T>(string path, out string msg) where T : ScriptableObject
        {
            if (ScaffoldExists(path)) { msg = $"already exists: {path}"; return true; }

            var dir = Path.GetDirectoryName(path)?.Replace('\\', '/');
            if (!string.IsNullOrEmpty(dir)) EnsureAssetFolder(dir);

            var so = ScriptableObject.CreateInstance<T>();
            AssetDatabase.CreateAsset(so, path);
            msg = path;
            return true;
        }

        // App Settings = PanelSettings со ссылкой на сгенерированную тему (1920x1080, scale-with-screen).
        private static bool CreatePanelSettings(out string msg)
        {
            var path = AppPaths.AppSettingsPath;
            if (ScaffoldExists(path)) { msg = $"already exists: {path}"; return true; }

            // тема нужна до PanelSettings — создаём, если ещё нет
            if (!ScaffoldExists(AppPaths.AppThemePath))
                WriteTextAssetIfMissing(AppPaths.AppThemePath, AppThemeContent(), out _);

            var dir = Path.GetDirectoryName(path)?.Replace('\\', '/');
            if (!string.IsNullOrEmpty(dir)) EnsureAssetFolder(dir);

            var ps = ScriptableObject.CreateInstance<PanelSettings>();
            ps.scaleMode = PanelScaleMode.ScaleWithScreenSize;
            ps.referenceResolution = new Vector2Int(1920, 1080);

            var theme = AssetDatabase.LoadAssetAtPath<ThemeStyleSheet>(AppPaths.AppThemePath);
            if (theme != null) ps.themeStyleSheet = theme;

            AssetDatabase.CreateAsset(ps, path);
            msg = theme != null ? path : $"{path} (theme not linked — AppTheme missing)";
            return true;
        }

        private void GenerateUids()
        {

        // Возвращает true, если файл записан. Пишет отчёт в log.
        private bool GenerateUids(StringBuilder log)
        {
            var pageIds  = Distinct(_pageReg.ReadStored());
            var popupIds = Distinct(_popupReg.ReadStored());

            if (pageIds.Count == 0 && popupIds.Count == 0)
            {
                log.AppendLine("UIDs: no page/popup ids in NavigationData — " + AppPaths.UidFileName + ".cs left untouched.");
                return false;
            }

            var body = new StringBuilder();
            body.AppendLine("// <auto-generated> Generated by App (Nexus) editor. Do not edit by hand. </auto-generated>");
            body.AppendLine("using Exerussus.AppCore.Navigation;");
            body.AppendLine();
            body.AppendLine($"namespace {AppPaths.GenNamespace}");
            body.AppendLine("{");
            body.AppendLine($"    public static partial class {AppPaths.UidFileName}");
            body.AppendLine("    {");

            var used = new HashSet<string>(StringComparer.Ordinal);
            int p = EmitUidBlock(body, used, pageIds,  "PageId",  "_Page");
            if (pageIds.Count > 0 && popupIds.Count > 0) body.AppendLine();
            int u = EmitUidBlock(body, used, popupIds, "PopupId", "_Popup");

            body.AppendLine("    }");
            body.AppendLine("}");

            if (!TryWrite(AppPaths.UidFileName, body.ToString(), out var path, out var err))
            {
                log.AppendLine("UIDs: write failed — " + err);
                return false;
            }

            log.AppendLine($"UIDs: {p} PageId, {u} PopupId -> {path}");
            return true;
        }

        // Пишет поля одного типа. id -> Pascal(id) + suffix; '_'/'-' в id отбрасываются.
        private static int EmitUidBlock(StringBuilder sb, HashSet<string> used,
                                        List<string> ids, string uidType, string suffix)
        {
            int n = 0;
            foreach (var id in ids)
            {
                var ident = Unique(used, ToTypeIdent(id) + suffix);
                var lit   = Escape(id);
                sb.AppendLine($"        public static readonly {uidType} {ident} = new {uidType}(\"{lit}\");");
                n++;
            }
            return n;
        }

        // ============================================================ запись кода

        private static bool TryWrite(string fileNameNoExt, string content, out string path, out string error)
        {
            path = $"{AppPaths.GenDir}/{fileNameNoExt}.cs";
            error = null;
            try
            {
                Directory.CreateDirectory(AppPaths.GenDir);
                File.WriteAllText(path, content, new UTF8Encoding(false));
                AssetDatabase.ImportAsset(path);
                return true;
            }
            catch (Exception e)
            {
                error = e.Message;
                return false;
            }
        }

        // ============================================================ именование / утилиты

        // id -> PascalCase, '_' и '-' (и прочие не-буквенно-цифровые) отбрасываются: game_over -> GameOver
        private static string ToTypeIdent(string raw)
        {
            var sb = new StringBuilder();
            bool up = true;
            foreach (var ch in raw ?? string.Empty)
            {
                if (char.IsLetterOrDigit(ch)) { sb.Append(up ? char.ToUpperInvariant(ch) : ch); up = false; }
                else up = true;
            }
            return EnsureValidIdent(sb.ToString());
        }

        private static string ToTypeIdentNonEmpty(string raw, string kind)
        {
            var s = ToTypeIdent(raw);
            return s == "_" || string.IsNullOrEmpty(s) ? "Unnamed" + kind : s;
        }

        private static string EnsureValidIdent(string s)
        {
            if (string.IsNullOrEmpty(s)) return "_";
            if (char.IsDigit(s[0])) s = "_" + s;
            if (CsKeywords.Contains(s)) s = "@" + s;
            return s;
        }

        private static string Unique(HashSet<string> used, string ident)
        {
            var baseId = ident;
            int i = 2;
            while (!used.Add(ident)) ident = baseId + "_" + i++;
            return ident;
        }

        private static List<string> Distinct(IEnumerable<string> src)
        {
            var seen = new HashSet<string>(StringComparer.Ordinal);
            var res  = new List<string>();
            if (src == null) return res;
            foreach (var raw in src)
            {
                var id = (raw ?? string.Empty).Trim();
                if (id.Length == 0) continue;
                if (seen.Add(id)) res.Add(id);
            }
            return res;
        }

        private static string Escape(string s) =>
            (s ?? string.Empty).Replace("\\", "\\\\").Replace("\"", "\\\"");

        private static readonly HashSet<string> CsKeywords = new(StringComparer.Ordinal)
        {
            "abstract","as","base","bool","break","byte","case","catch","char","checked","class","const",
            "continue","decimal","default","delegate","do","double","else","enum","event","explicit","extern",
            "false","finally","fixed","float","for","foreach","goto","if","implicit","in","int","interface",
            "internal","is","lock","long","namespace","new","null","object","operator","out","override","params",
            "private","protected","public","readonly","ref","return","sbyte","sealed","short","sizeof","stackalloc",
            "static","string","struct","switch","this","throw","true","try","typeof","uint","ulong","unchecked",
            "unsafe","ushort","using","virtual","void","volatile","while"
        };
    }
