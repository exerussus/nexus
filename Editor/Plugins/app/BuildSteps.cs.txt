// ============================================================
//  BuildSteps.cs — поиск шагов сборки и хост их UI.
//
//  Контракт шагов (BuildStep<TSettings>, [BuildStep], BuildStepContext, BuildTokens,
//  IBuildStepHost) живёт в AppCore: Exerussus.AppCore.Editor.Build. Страница находит
//  шаги во ВСЕХ сборках проекта (TypeCache), поэтому шаг может прийти из любого пакета,
//  который ссылается на Exerussus.AppCore.Editor: нет пакета — нет шага.
//
//  Свои шаги страницы (zip, version.json, S3) — в соседних файлах; им доступен AppKit
//  через AppStepHost.Kit(host).
// ============================================================

using System;
using System.Collections.Generic;
using Exerussus.AppCore.Editor.Build;
using Exerussus.Nexus.Abstractions;
using UnityEditor;
using UnityEngine;
using UnityEngine.UIElements;

namespace Exerussus.Nexus.Pages.App
{
    /// <summary>Найденный шаг: метаданные + экземпляр.</summary>
    internal sealed class StepDescriptor
    {
        public BuildStepAttribute Meta;
        public IBuildStep         Step;

        /// <summary>Шаг не из страницы App (пришёл с пакетом).</summary>
        public bool External;

        /// <summary>Сборка, из которой пришёл шаг (для подсказки в UI).</summary>
        public string Assembly = "";
    }

    internal static class BuildStepRegistry
    {
        /// <summary>Все шаги проекта (без кэша — TypeCache и так быстрый).</summary>
        public static List<StepDescriptor> All()
        {
            var list = new List<StepDescriptor>();
            var own  = typeof(BuildStepRegistry).Assembly;
            foreach (var type in TypeCache.GetTypesWithAttribute<BuildStepAttribute>())
            {
                if (type.IsAbstract || !typeof(IBuildStep).IsAssignableFrom(type)) continue;
                var meta = (BuildStepAttribute)Attribute.GetCustomAttribute(type, typeof(BuildStepAttribute));
                if (meta == null || string.IsNullOrEmpty(meta.Id)) continue;
                if (Find(list, meta.Id) is { } dup)
                {
                    NexusDiagnostics.Error("Build", $"id шага «{meta.Id}» занят дважды — {dup.Step.GetType().FullName} и {type.FullName}; второй пропущен");
                    continue;
                }
                try
                {
                    list.Add(new StepDescriptor
                    {
                        Meta     = meta,
                        Step     = (IBuildStep)Activator.CreateInstance(type),
                        External = type.Assembly != own,
                        Assembly = type.Assembly.GetName().Name,
                    });
                }
                catch (Exception ex)
                {
                    NexusDiagnostics.Error($"Build: шаг {type.FullName} не создаётся", ex);
                }
            }
            list.Sort((a, b) => string.CompareOrdinal(a.Meta.Id, b.Meta.Id));
            return list;
        }

        public static StepDescriptor Find(List<StepDescriptor> all, string id)
        {
            foreach (var d in all)
                if (d.Meta.Id == id) return d;
            return null;
        }

        /// <summary>
        /// Синхронизировать пайплайн профиля с найденными шагами: новые шаги
        /// дописываются выключенными в конец. Неизвестные записи НЕ удаляются
        /// (шаг мог временно пропасть) — их показывает UI. true — были изменения.
        /// </summary>
        public static bool Sync(ProfilePipeline pipeline, List<StepDescriptor> all)
        {
            var changed = false;
            foreach (var d in all)
            {
                var exists = false;
                foreach (var e in pipeline.steps)
                    if (e.id == d.Meta.Id) { exists = true; break; }
                if (exists) continue;
                pipeline.steps.Add(new StepEntry { id = d.Meta.Id, enabled = false });
                changed = true;
            }
            return changed;
        }
    }

    /// <summary>IBuildStepHost поверх AppKit: шаги из пакетов рисуются в стиле страницы.</summary>
    internal sealed class AppStepHost : IBuildStepHost
    {
        public readonly AppKit AppKit;

        public AppStepHost(AppKit kit) => AppKit = kit;

        /// <summary>AppKit для шагов самой страницы.</summary>
        public static AppKit Kit(IBuildStepHost host)
            => (host as AppStepHost)?.AppKit ?? throw new InvalidOperationException("Шаг страницы App показан не страницей App");

        public string UserConfigPath(string file) => AppKit.Context.GetUserConfigPath(file);

        public void SetStatus(string text, BuildStepStatus status = BuildStepStatus.Info)
            => AppKit.Context.SetStatus(text, status switch
            {
                BuildStepStatus.Ok      => StatusKind.Ok,
                BuildStepStatus.Warning => StatusKind.Warning,
                BuildStepStatus.Error   => StatusKind.Error,
                _                       => StatusKind.Info,
            });

        public TextField Field(string value, Action<string> onChanged, bool password = false)
        {
            var f = AppKit.Input(value ?? "", onChanged);
            f.isDelayed = true;
            if (password) f.isPasswordField = true;
            f.style.flexGrow = 1f;
            return f;
        }

        public Toggle Toggle(string label, bool value, Action<bool> onChanged) => AppKit.Check(label, value, onChanged);

        public Button Button(string text, Action onClick, bool primary = false)
            => AppKit.Btn(primary ? NexusRole.Primary : NexusRole.Common, text, onClick, small: true);

        public VisualElement Labeled(string label, VisualElement field) => AppKit.Labeled(label, field, 130f);

        public Label Hint(string text) => AppKit.Hint(text);

        public Label Section(string text)
        {
            var l = AppKit.Text(text, NexusToken.TextDim, 11);
            l.style.unityFontStyleAndWeight = FontStyle.Bold;
            l.style.marginTop = 8f;
            l.style.marginBottom = 2f;
            return l;
        }
    }
}
