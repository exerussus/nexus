// ============================================================
//  AppPage.cs — страница Nexus «App»: всё, что обслуживает AppCore в проекте.
//
//  Обслуживающая (servicing) страница: её продукт живёт в Assets/App (папка целиком
//  принадлежит AppCore), потребляется игрой напрямую и переживает выключение страницы.
//
//  Вкладки — отдельные классы (IAppTab), страница только переключает их:
//    Editor      — реестр id страниц/попапов/фрагментов и навигационные классы;
//    Generation  — NavTargets, вёрстка вью по канону, скаффолд Assets/App;
//    Nav classes — стандартные навигационные классы;
//    External    — внешний след страницы с путями и очисткой сгенерированного;
//    Text        — наборы символов, цепочка шрифтов, запекание;
//    Сборка      — билд, пайплайн, BuildInfo и строка версии (бывшая страница Build).
// ============================================================

using System;
using System.Collections.Generic;
using Exerussus.AppCore;
using Exerussus.AppCore.Editor;
using Exerussus.Nexus.Abstractions;
using UnityEditor;
using UnityEngine;
using UnityEngine.UIElements;

namespace Exerussus.Nexus.Pages.App
{
    /// <summary>Вкладка страницы App.</summary>
    internal interface IAppTab
    {
        string Title { get; }

        /// <summary>Строит корень вкладки один раз.</summary>
        VisualElement Build();

        /// <summary>Вкладка стала видимой: перечитать данные и перерисоваться.</summary>
        void Show();

        /// <summary>Вкладка скрыта: сбросить несохранённое.</summary>
        void Hide();

        /// <summary>Явный Refresh из тулбара хаба.</summary>
        void Refresh();

        /// <summary>Можно ли закрыть страницу (нет неприменённых правок, не идёт сборка).</summary>
        bool CanClose();

        void Dispose();
    }

    [EditorPage("app", "App", PageCategory.Infrastructure, order: 10, icon: "d_Linked")]
    public sealed class AppPage : EditorPageBase
    {
        private const string TabKey = "tab";

        private AppKit _kit;
        private readonly List<IAppTab> _tabs = new();
        private readonly List<VisualElement> _tabRoots = new();
        private VisualElement _root;
        private VisualElement _headerStatus;
        private int _current = -1;

        protected override void OnInitialize()
        {
            _kit = new AppKit(Context);

            _tabs.Add(new IdsTab(_kit));
            _tabs.Add(new GenerationTab(_kit));
            _tabs.Add(new NavClassesTab(_kit));
            _tabs.Add(new ExternalTab(_kit));
            _tabs.Add(new TextTab(_kit));
            _tabs.Add(new BuildTab(_kit));
        }

        public override VisualElement BuildUI()
        {
            _root = new VisualElement();
            _root.style.flexGrow = 1;
            _root.style.backgroundColor = _kit.Tok(NexusToken.BgHard);
            _root.style.paddingLeft = 20f; _root.style.paddingRight = 20f;
            _root.style.paddingTop = 16f;

            // ---- шапка ----
            var header = _kit.Row();
            header.style.flexWrap = Wrap.NoWrap;
            header.style.marginBottom = 10f;

            var title = _kit.Text("App", NexusToken.TextNormal, 16);
            title.style.unityFontStyleAndWeight = FontStyle.Bold;
            title.style.marginRight = 12f;
            header.Add(title);
            header.Add(_kit.Mono("com.exerussus.app-core 4.0", NexusToken.TextFaint));
            header.Add(_kit.Spacer());

            _headerStatus = new VisualElement();
            header.Add(_headerStatus);
            _root.Add(header);

            // ---- вкладки ----
            var titles = new string[_tabs.Count];
            for (var i = 0; i < _tabs.Count; i++) titles[i] = _tabs[i].Title;

            var selected = Mathf.Clamp(SessionState.GetInt(Context.GetSessionKey(TabKey), 0), 0, _tabs.Count - 1);
            var tabBar = Context.Ui.Tabs(titles, selected, Select);
            tabBar.style.marginBottom = 14f;
            _root.Add(tabBar);

            foreach (var tab in _tabs)
            {
                var scroll = new ScrollView(ScrollViewMode.Vertical);
                scroll.style.flexGrow = 1;
                scroll.contentContainer.style.paddingBottom = 20f;
                scroll.Add(tab.Build());
                scroll.style.display = DisplayStyle.None;
                _tabRoots.Add(scroll);
                _root.Add(scroll);
            }

            _current = selected;
            _tabRoots[_current].style.display = DisplayStyle.Flex;
            return _root;
        }

        private void Select(int index)
        {
            if (index == _current || index < 0 || index >= _tabs.Count) return;

            if (_current >= 0)
            {
                _tabs[_current].Hide();
                _tabRoots[_current].style.display = DisplayStyle.None;
            }

            _current = index;
            SessionState.SetInt(Context.GetSessionKey(TabKey), index);
            _tabRoots[index].style.display = DisplayStyle.Flex;
            _tabs[index].Show();
            RefreshHeader();
        }

        private void RefreshHeader()
        {
            if (_headerStatus == null) return;
            _headerStatus.Clear();

            var settings = AppCoreAssets.FindNavigationSettings();
            _headerStatus.Add(settings != null
                ? _kit.Badge(StatusKind.Ok, "NavigationSettings на месте")
                : _kit.Badge(StatusKind.Warning, $"Нет {AppCorePaths.NavigationSettingsPath}"));
        }

        public override void OnEnter()
        {
            RefreshHeader();
            if (_current >= 0) _tabs[_current].Show();
        }

        public override void OnExit()
        {
            if (_current >= 0) _tabs[_current].Hide();
        }

        public override void OnRefresh()
        {
            RefreshHeader();
            if (_current >= 0) _tabs[_current].Refresh();
        }

        public override bool CanClose()
        {
            foreach (var tab in _tabs)
                if (!tab.CanClose()) return false;
            return true;
        }

        public override void OnPrePack()
        {
            foreach (var tab in _tabs) tab.Hide();
        }

        protected override void OnDisposing()
        {
            foreach (var tab in _tabs)
            {
                try { tab.Dispose(); }
                catch (Exception e) { Debug.LogException(e); }
            }
        }
    }
}
